from decimal import Decimal

from django.conf import settings
from django.db import models
from django.utils import timezone


class WenzeProduct(models.Model):
    STATUS_CHOICES = (
        ("DRAFT", "Brouillon"),
        ("ACTIVE", "Actif"),
        ("PAUSED", "Suspendu"),
        ("ARCHIVED", "Archivé"),
    )

    seller = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.CASCADE,
        related_name="wenze_products",
    )
    name = models.CharField(max_length=255)
    description = models.TextField(blank=True)
    price = models.DecimalField(max_digits=14, decimal_places=2)
    currency = models.CharField(max_length=3, default="EUR")
    stock = models.PositiveIntegerField(default=0)
    status = models.CharField(
        max_length=20,
        choices=STATUS_CHOICES,
        default="DRAFT",
    )
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ["-created_at"]

    def __str__(self):
        return self.name

    @property
    def active_buka_prix(self):
        now = timezone.now()
        return self.buka_prix_campaigns.filter(
            active=True,
            start_at__lte=now,
            end_at__gte=now,
        ).order_by("-created_at").first()

    @property
    def current_price(self):
        campaign = self.active_buka_prix
        if campaign:
            return campaign.promotion_price
        return self.price

class BukaPrixCampaign(models.Model):
    product = models.ForeignKey(
        WenzeProduct,
        on_delete=models.CASCADE,
        related_name="buka_prix_campaigns",
    )
    title = models.CharField(max_length=255)
    promotion_price = models.DecimalField(
        max_digits=14,
        decimal_places=2,
    )
    start_at = models.DateTimeField()
    end_at = models.DateTimeField()
    active = models.BooleanField(default=False)
    created_by = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="buka_prix_campaigns_created",
    )
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ["-created_at"]

    def __str__(self):
        return self.title

    @property
    def is_running(self):
        now = timezone.now()
        return (
            self.active
            and self.start_at <= now <= self.end_at
        )

    def save(self, *args, **kwargs):
        if self.promotion_price < Decimal("0.00"):
            self.promotion_price = Decimal("0.00")
        super().save(*args, **kwargs)

from decimal import Decimal
from uuid import uuid4

from django.contrib.auth.decorators import login_required
from django.db import transaction
from django.http import HttpResponse
from django.shortcuts import get_object_or_404, render

from .models import BukaPrixCampaign, WenzeOrder, WenzeProduct


def index(request):
    products = WenzeProduct.objects.filter(status="ACTIVE").select_related("seller")
    campaigns = BukaPrixCampaign.objects.filter(active=True).select_related("product")

    return render(
        request,
        "wenze/index.html",
        {
            "products": products,
            "campaigns": campaigns,
        },
    )


@login_required
@transaction.atomic
def create_order(request, product_id):
    product = get_object_or_404(
        WenzeProduct.objects.select_for_update(),
        id=product_id,
        status="ACTIVE",
    )

    if request.method != "POST":
        return HttpResponse("Méthode non autorisée", status=405)

    quantity = int(request.POST.get("quantity", "1"))

    if quantity < 1:
        return HttpResponse("Quantité invalide", status=400)

    if product.stock < quantity:
        return HttpResponse("Stock insuffisant", status=400)

    unit_price = product.current_price
    total = unit_price * Decimal(quantity)

    order = WenzeOrder.objects.create(
        buyer=request.user,
        product=product,
        quantity=quantity,
        unit_price=unit_price,
        currency=product.currency,
        total_amount=total,
        reference=f"WENZE-{uuid4().hex[:16].upper()}",
    )

    product.stock -= quantity
    product.save(update_fields=["stock", "updated_at"])

    return render(
        request,
        "wenze/order_created.html",
        {"order": order},
    )
